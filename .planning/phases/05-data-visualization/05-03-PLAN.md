- **Floating mode drag** (pointer events + CSS transform, matching SimCanvas/ForceGraph pattern):
  ```typescript
  const posRef = useRef({ x: 100, y: 100 });
  const dragStartRef = useRef({ x: 0, y: 0, ox: 0, oy: 0 });
  const panelRef = useRef<HTMLDivElement>(null);

  const onPointerDown = (e: React.PointerEvent) => {
    (e.target as HTMLElement).setPointerCapture(e.pointerId);
    dragStartRef.current = {
      x: e.clientX, y: e.clientY,
      ox: posRef.current.x, oy: posRef.current.y,
    };
    e.preventDefault(); // prevent scroll on touch (Pitfall 5)
  };
  const onPointerMove = (e: React.PointerEvent) => {
    if (!(e.target as HTMLElement).hasPointerCapture(e.pointerId)) return;
    posRef.current = {
      x: dragStartRef.current.ox + (e.clientX - dragStartRef.current.x),
      y: dragStartRef.current.oy + (e.clientY - dragStartRef.current.y),
    };
    if (panelRef.current) {
      panelRef.current.style.transform =
        `translate(${posRef.current.x}px, ${posRef.current.y}px)`;
    }
  };
  const onPointerUp = (e: React.PointerEvent) => {
    (e.target as HTMLElement).releasePointerCapture(e.pointerId);
  };
  ```
  - Drag handle: only the header bar in floating mode, with `data-testid="drag-handle"`, `cursor: grab`, `touch-action: none`
  - Pointer event handlers only attached to drag handle element (not entire panel)
