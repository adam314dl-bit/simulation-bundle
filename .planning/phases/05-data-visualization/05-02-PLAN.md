- **Export `bilinearSample` as a named function** (for testability):
  ```typescript
  export function bilinearSample(
    data: Float64Array, w: number, h: number, x: number, y: number
  ): number {
    const x0 = Math.floor(x), y0 = Math.floor(y);
    const x1 = Math.min(x0 + 1, w - 1), y1 = Math.min(y0 + 1, h - 1);
    const fx = x - x0, fy = y - y0;
    const v00 = data[y0 * w + x0]!;
    const v10 = data[y0 * w + x1]!;
    const v01 = data[y1 * w + x0]!;
    const v11 = data[y1 * w + x1]!;
    return v00 * (1 - fx) * (1 - fy) + v10 * fx * (1 - fy) +
           v01 * (1 - fx) * fy + v11 * fx * fy;
  }
  ```

- Component uses `useRef<HTMLCanvasElement>` and `useEffect` to render on data change

- Rendering logic in `useEffect([data, gridWidth, gridHeight, colorRamp, opacity, interpolate, range])`:
  ```typescript
  const canvas = canvasRef.current;
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  if (!ctx) return;
  const lut = getRampLUT(colorRamp ?? 'viridis');
  const [rangeMin, rangeMax] = range ?? [0, 1];
  const rangeSpan = rangeMax - rangeMin || 1;
  const cw = canvas.width, ch = canvas.height;
  const imageData = ctx.createImageData(cw, ch);
  const pixels = imageData.data;

  for (let py = 0; py < ch; py++) {
    for (let px = 0; px < cw; px++) {
      const gx = (px / cw) * (gridWidth - 1);
      const gy = (py / ch) * (gridHeight - 1);
      const raw = interpolate !== false
        ? bilinearSample(data, gridWidth, gridHeight, gx, gy)
        : data[Math.round(gy) * gridWidth + Math.round(gx)]!;
      const t = Math.max(0, Math.min(1, (raw - rangeMin) / rangeSpan));
      const lutIdx = Math.round(t * 255) * 4;
      const pIdx = (py * cw + px) * 4;
      pixels[pIdx] = lut[lutIdx]!;
      pixels[pIdx + 1] = lut[lutIdx + 1]!;
      pixels[pIdx + 2] = lut[lutIdx + 2]!;
      pixels[pIdx + 3] = Math.round((opacity ?? 0.7) * 255);
    }
  }
  ctx.putImageData(imageData, 0, 0);
  ```
